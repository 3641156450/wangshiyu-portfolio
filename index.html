tool.js · 交互裁剪 + 5 寸排版 + 下载
var CROP_W=360, CROP_H=500;      // 交互画布
var st = { zoom:1, offX:0, offY:0 }; // 照片缩放 + 偏移

// 裁剪框：居中，尺寸按证件照规格比例
function cropBox(){
  var bw=280, bh=Math.round(bw*spec.h/spec.w);
  return { x:(CROP_W-bw)/2, y:(CROP_H-bh)/2, w:bw, h:bh };
}
// 边界限制：照片必须完全盖住裁剪框（不露白边）
function clampOffset(){
  var b=cropBox(), dw=img.width*st.zoom, dh=img.height*st.zoom;
  var cx=CROP_W/2+st.offX, cy=CROP_H/2+st.offY;
  cx=Math.min(Math.max(cx, b.x+b.w-dw), b.x);   // 左不越界 / 右不露白
  cy=Math.min(Math.max(cy, b.y+b.h-dh), b.y);
  st.offX=cx-CROP_W/2; st.offY=cy-CROP_H/2;
}
// 缩放：围绕照片中心放大缩小，再限制边界
function setZoom(nz){
  st.offX += img.width*(st.zoom-nz)/2;
  st.offY += img.height*(st.zoom-nz)/2;
  st.zoom = nz; clampOffset(); render();
}

// 渲染：把裁剪框区域映射回照片原坐标 → 单张 → 5 寸排版
function render(){
  var b = cropBox();
  var sx=(b.x-CROP_W/2-st.offX)/st.zoom; // 源图裁剪区
  var sy=(b.y-CROP_H/2-st.offY)/st.zoom;
  cropCtx.drawImage(img, sx, sy, b.w/st.zoom, b.h/st.zoom,
                    0, 0, spec.w, spec.h); // 输出证件照
  layoutSheet(cropCanvas, spec); // 5 寸排版（拖动/缩放后自动更新）
  drawInteract();                 // 重绘交互画布（照片+遮罩+参考线）
}

// 拖动与滚轮交互
canvas.addEventListener("mousedown", function(e){ drag={x:e.clientX,y:e.clientY}; });
window.addEventListener("mousemove", function(e){
  if(!drag) return;
  st.offX+=(e.clientX-drag.x)*scale; st.offY+=(e.clientY-drag.y)*scale; // 拖动照片
  drag={x:e.clientX,y:e.clientY}; clampOffset(); render();
});
window.addEventListener("mouseup", function(){ drag=null; });
canvas.addEventListener("wheel", function(e){ // 滚轮缩放
  e.preventDefault();
  setZoom(st.zoom * (e.deltaY<0 ? 1.1 : 1/1.1));
});

// 下载排版图：PNG / JPG 可选格式
function downloadSheet(fmt){
  var mime = fmt=="jpg" ? "image/jpeg" : "image/png";
  var ext  = fmt=="jpg" ? "jpg" : "png";
  sheetCanvas.toBlob(function(blob){           // 画布导出二进制
    var a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "证件照5寸排版." + ext;   // 按格式命名
    a.click();
  }, mime, fmt=="jpg" ? 0.92 : undefined); // JPG 指定画质
}
